import os
import subprocess
import json
from datetime import datetime
import sys
import io

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
sys.stderr = io.TextIOWrapper(sys.stderr.buffer, encoding="utf-8")

VIDEOS_DIR = "videos"
IDEAS_FILE = "ideas.json"

def load_ideas():
    with open(IDEAS_FILE, "r", encoding="utf-8") as f:
        return json.load(f)

def save_ideas(data):
    with open(IDEAS_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)

def get_next_video_id():
    os.makedirs(VIDEOS_DIR, exist_ok=True)
    existing = [d for d in os.listdir(VIDEOS_DIR) if d.startswith("video_")]

    if not existing:
        return 1

    nums = [int(d.split("_")[1]) for d in existing if d.split("_")[1].isdigit()]
    return max(nums) + 1

def build_script_text(script_obj):
    return (
        f"{script_obj['hook']} "
        f"{script_obj['beat']} "
        f"{script_obj['lore']} "
        f"{script_obj['cta']}"
    )

def get_next_pending_idea_key(ideas):
    for key, idea in ideas.items():
        if idea.get("status") == "pending":
            return key
    return None

def create_video_job(video_index, idea_key):
    video_id = f"video_{video_index:03d}"
    base_path = os.path.join(VIDEOS_DIR, video_id)

    input_dir = os.path.join(base_path, "input")
    intermediate_dir = os.path.join(base_path, "intermediate")
    output_dir = os.path.join(base_path, "output")

    os.makedirs(input_dir, exist_ok=True)
    os.makedirs(intermediate_dir, exist_ok=True)
    os.makedirs(output_dir, exist_ok=True)

    print(f"\n🎬 Creating {video_id} from {idea_key}")

    ideas = load_ideas()
    idea = ideas[idea_key]

    frozen_idea_path = os.path.join(input_dir, "idea.json")
    with open(frozen_idea_path, "w", encoding="utf-8") as f:
        json.dump({idea_key: idea}, f, indent=2)

    print("🧊 Frozen idea snapshot:", frozen_idea_path)

    # Script
    script_text = build_script_text(idea["script"])
    script_path = os.path.join(input_dir, "script.txt")

    with open(script_path, "w", encoding="utf-8") as f:
        f.write(script_text)

    # TTS
    subprocess.run(
        [sys.executable, "scripts/narrate_google.py", script_path, intermediate_dir],
        check=True
    )

    narration_path = os.path.join(intermediate_dir, "narration.wav")

    # Whisper
    words_json = os.path.join(intermediate_dir, "words.json")
    subprocess.run(
        [sys.executable, "alignment/whisper_align.py", narration_path, words_json],
        check=True
    )

    # Segments
    segments_json = os.path.join(intermediate_dir, "segment_timeline.json")
    subprocess.run(
        [sys.executable, "alignment/segment_mapper.py",
         words_json, frozen_idea_path, segments_json],
        check=True
    )

    # Captions
    blocks_json = os.path.join(intermediate_dir, "word_blocks.json")
    subprocess.run(
        [sys.executable, "captions/build_word_blocks.py",
         words_json, frozen_idea_path, idea_key, blocks_json],
        check=True
    )

    # Audio
    final_audio = os.path.join(output_dir, "audio.wav")
    subprocess.run(
        [sys.executable, "audio/audio_post.py",
         narration_path, words_json, segments_json,
         frozen_idea_path, final_audio],
        check=True
    )

    # Video
    final_video = os.path.join(output_dir, "video.mp4")
    subprocess.run(
        [sys.executable, "render/render.py",
         final_audio, blocks_json, final_video, idea_key],
        check=True
    )

    # Upload
    title = idea["youtube"]["title"]
    description = idea["youtube"]["description"]
    tags = ",".join(idea["youtube"]["tags"])

    result = subprocess.run(
        [
            sys.executable,
            "scripts/youtube_uploader.py",
            final_video,
            title,
            description,
            tags
        ]
    )

    if result.returncode == 99:
        print("⏸ Batch stopped due to YouTube upload limit.")
        print("🛑 Idea NOT marked rendered. Resume later.")
        sys.exit(0)

    if result.returncode != 0:
        raise RuntimeError("❌ Upload failed unexpectedly.")

    # Update ideas.json
    ideas = load_ideas()
    ideas[idea_key]["status"] = "rendered"
    ideas[idea_key]["log"]["rendered_at"] = datetime.now().isoformat()
    ideas[idea_key]["log"]["video_path"] = final_video
    save_ideas(ideas)

    print(f"✅ {video_id} completed and {idea_key} uploaded + scheduled")

def generate_batch(count=1):
    ideas = load_ideas()
    start_id = get_next_video_id()

    for i in range(count):
        idea_key = get_next_pending_idea_key(ideas)

        if not idea_key:
            print("🚫 No pending ideas left")
            return

        create_video_job(start_id + i, idea_key)

        # Mark consumed in memory
        ideas[idea_key]["status"] = "rendered"

if __name__ == "__main__":
    count = int(sys.argv[1]) if len(sys.argv) > 1 else 1
    generate_batch(count)

import sys
import os
import re
from google.cloud import texttospeech

# ---------- USAGE ----------
# python narrate_google.py <script_path> <job_intermediate_dir>

if len(sys.argv) < 3:
    print("Usage: python narrate_google.py <script_path> <job_intermediate_dir>")
    sys.exit(1)

SCRIPT_PATH = sys.argv[1]
JOB_DIR = sys.argv[2]

FINAL_AUDIO = os.path.join(JOB_DIR, "narration.wav")
os.makedirs(JOB_DIR, exist_ok=True)

# ---------- LOAD SCRIPT ----------
with open(SCRIPT_PATH, "r", encoding="utf-8") as f:
    text = f.read().strip()

# ---------- CHANNEL NAME PRONUNCIATION FIX ----------
# Convert Fact4Sigma → Fact for Sigma (TTS-friendly)
text = re.sub(r"\bFact4Sigma\b", "Fact for Sigma", text, flags=re.IGNORECASE)

print("🎙 Using Google Cloud TTS")

# ---------- INIT CLIENT ----------
client = texttospeech.TextToSpeechClient()

# ---------- BUILD VOICE ----------
voice = texttospeech.VoiceSelectionParams(
    language_code="en-US",
    name="en-US-Standard-D"
)

# ---------- AUDIO CONFIG ----------
audio_config = texttospeech.AudioConfig(
    audio_encoding=texttospeech.AudioEncoding.LINEAR16,
    speaking_rate=1.0,
    pitch=0.5,
    volume_gain_db=2.0
)

# ---------- INPUT ----------
synthesis_input = texttospeech.SynthesisInput(text=text)

MAX_CHARS = 3000
if len(text) > MAX_CHARS:
    raise RuntimeError(f"❌ Script too long: {len(text)} chars (limit {MAX_CHARS})")

# ---------- CALL API ----------
response = client.synthesize_speech(
    input=synthesis_input,
    voice=voice,
    audio_config=audio_config
)

# ---------- SAVE ----------
with open(FINAL_AUDIO, "wb") as out:
    out.write(response.audio_content)

print(f"✅ Narration built: {FINAL_AUDIO}")

import sys
import json
import whisper
import string

if len(sys.argv) < 3:
    print("Usage: python whisper_align.py <input_wav> <output_json>")
    sys.exit(1)

audio_path = sys.argv[1]
output_path = sys.argv[2]

print("🎧 Loading Whisper model...")
model = whisper.load_model("base")

print("🕒 Running alignment...")
result = model.transcribe(audio_path, word_timestamps=True)

raw_words = []
for segment in result["segments"]:
    for w in segment["words"]:
        raw_words.append({
            "word": w["word"],
            "start": round(w["start"], 3),
            "end": round(w["end"], 3)
        })

def normalize_token(token):
    return token.strip().lower().translate(str.maketrans("", "", string.punctuation))

words = []
i = 0

while i < len(raw_words):
    w1 = raw_words[i]
    t1 = normalize_token(w1["word"])

    # Look ahead for "fact for sigma" or "fact four sigma"
    if i + 2 < len(raw_words):
        w2 = raw_words[i + 1]
        w3 = raw_words[i + 2]

        t2 = normalize_token(w2["word"])
        t3 = normalize_token(w3["word"])

        if t1 == "fact" and t2 in ("for", "four") and t3 == "sigma":
            words.append({
                "word": "FACT4SIGMA",
                "start": w1["start"],
                "end": w3["end"]
            })
            i += 3
            continue

    # Default: pass through original word
    words.append({
        "word": w1["word"],
        "start": w1["start"],
        "end": w1["end"]
    })
    i += 1

with open(output_path, "w", encoding="utf-8") as f:
    json.dump(words, f, indent=2)

print(f"✅ Word alignment written to: {output_path}")

alignment/segment_mapper.py

import sys
import json
import re

# ---------------- NORMALIZATION ----------------

def normalize_token(token: str) -> str:
    return re.sub(r"[^a-z0-9]", "", token.lower())

STOPWORDS = {
    "the", "a", "an", "and", "or", "but",
    "is", "are", "was", "were", "be",
    "to", "of", "in", "on", "for", "with",
    "there", "this", "that", "it",
    "you", "we", "they", "i",
    "do", "does", "did", "not"
}

def is_strong(token: str) -> bool:
    return token and token not in STOPWORDS

# ---------------- ARGUMENTS ----------------

if len(sys.argv) < 4:
    print("Usage: python segment_mapper.py <words.json> <idea.json> <output.json>")
    sys.exit(1)

WORDS_PATH = sys.argv[1]
IDEA_PATH = sys.argv[2]
OUTPUT_PATH = sys.argv[3]

# ---------------- LOAD INPUTS ----------------

with open(WORDS_PATH, "r", encoding="utf-8") as f:
    raw_whisper_words = json.load(f)

with open(IDEA_PATH, "r", encoding="utf-8") as f:
    ideas = json.load(f)

# 🔒 SINGLE IDEA (FROZEN)
idea = next(iter(ideas.values()))

script = idea["script"]

segments = [
    ("hook", script["hook"]),
    ("beat", script["beat"]),
    ("lore", script["lore"]),
    ("cta", script["cta"]),
]

# ---------------- NORMALIZE WHISPER ----------------

whisper_seq = []
for w in raw_whisper_words:
    token = normalize_token(w["word"])
    if token:
        whisper_seq.append({
            "word": token,
            "start": w["start"],
            "end": w["end"],
        })

if not whisper_seq:
    raise RuntimeError("No usable Whisper words")

# ---------------- ALIGN SEGMENTS ----------------

cursor = 0
timeline = {}

for idx, (name, text) in enumerate(segments):
    raw_tokens = [normalize_token(t) for t in text.split()]
    tokens = [t for t in raw_tokens if is_strong(t)]

    matched_times = []

    for token in tokens:
        while cursor < len(whisper_seq):
            w = whisper_seq[cursor]
            if w["word"] == token:
                matched_times.append((w["start"], w["end"]))
                cursor += 1
                break
            cursor += 1

    if len(matched_times) >= 2:
        start_time = matched_times[0][0]
        end_time = matched_times[-1][1]

    elif len(matched_times) == 1:
        center_start, center_end = matched_times[0]
        start_time = max(0, center_start - 0.35)
        end_time = center_end + 0.35

    else:
        if idx > 0:
            start_time = timeline[segments[idx - 1][0]]["end"]
        else:
            start_time = whisper_seq[0]["start"]

        if idx + 1 < len(segments):
            end_time = whisper_seq[min(len(whisper_seq) - 1, cursor)]["end"]
        else:
            end_time = whisper_seq[-1]["end"]

    timeline[name] = {
        "start": round(start_time, 3),
        "end": round(end_time, 3),
    }

# ---------------- WRITE OUTPUT ----------------

with open(OUTPUT_PATH, "w", encoding="utf-8") as f:
    json.dump({"segments": timeline}, f, indent=2)

print("✅ Segment timeline built:")
for k, v in timeline.items():
    print(f"  {k}: {v['start']} → {v['end']}")

import json
import os
import sys

if len(sys.argv) < 5:
    print("Usage: python build_word_blocks.py <words.json> <ideas.json> <idea_key> <output.json>")
    sys.exit(1)

WORDS_PATH = sys.argv[1]

MAX_WORDS = 3
MAX_DURATION = 0.8
PAUSE_THRESHOLD = 0.18
MAX_CHARS = 18

# ---------------- HELPERS ----------------

def normalize(w: str) -> str:
    return w.strip().lower().strip(".,!?\"'")

# ---------------- LOAD INPUTS ----------------

with open(WORDS_PATH, "r", encoding="utf-8") as f:
    words = json.load(f)

with open(IDEAS_PATH, "r", encoding="utf-8") as f:
    ideas = json.load(f)

from pydub import AudioSegment
import json
import sys
import os
import string

ASSET_REGISTRY = "assets_registry.json"

# ------------------- HELPERS -------------------

def load_json(path):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def normalize_word(w):
    return w.strip().lower().translate(str.maketrans("", "", string.punctuation))

def find_word_time(words, target):
    target = normalize_word(target)
    for w in words:
        if normalize_word(w["word"]) == target:
            return w["start"]
    return None

def find_segment_time(segment_timeline, name):
    name = name.lower()
    segment = segment_timeline.get(name)
    if not segment:
        return None
    return segment["start"]

# ------------------- CORE ENGINE -------------------

def apply_music(narration, music_cfg, registry):
    music_id = music_cfg["asset"]
    music_asset = registry["music"].get(music_id)

    if not music_asset:
        print("⚠️ Music not found in registry:", music_id)
        return narration

    music = AudioSegment.from_file(music_asset["path"])
    loops = int(len(narration) / len(music)) + 1
    music = (music * loops)[:len(narration)]

    volume = music_cfg.get("overrides", {}).get(
        "volume", music_asset.get("default_volume", -10)
    )
    music = music + volume

    if music_asset.get("duck_under_voice", False):
        music = music - 10

    return narration.overlay(music)

def apply_sfx(base_audio, sfx_list, words, segment_timeline, registry):
    for sfx in sfx_list:
        sfx_id = sfx["asset"]
        sfx_asset = registry["sfx"].get(sfx_id)

        if not sfx_asset:
            print("⚠️ Missing SFX in registry:", sfx_id)
            continue

        sound = AudioSegment.from_file(sfx_asset["path"])
        volume = sfx.get("overrides", {}).get(
            "volume", sfx_asset.get("default_volume", 0)
        )
        sound = sound + volume

        start_time = None

        if "at_word" in sfx:
            start_time = find_word_time(words, sfx["at_word"])
        elif "at_segment" in sfx:
            start_time = find_segment_time(segment_timeline, sfx["at_segment"])
        elif "start" in sfx:
            start_time = sfx["start"]

        if start_time is None:
            continue

        base_audio = base_audio.overlay(
            sound,
            position=int(start_time * 1000)
        )

    return base_audio

# ------------------- PIPELINE -------------------

def render_audio_with_idea(input_audio, words_json, segments_json, idea_json_path, output_audio):

import os
import json
import sys
import random

import numpy as np
from PIL import Image, ImageDraw, ImageFont

from moviepy import (
    VideoFileClip,
    AudioFileClip,
    CompositeVideoClip,
    ImageClip,
)

from overlay_engine import build_overlay_clips

# ---------------- CONFIG ----------------
WIDTH = 1080
HEIGHT = 1920
FPS = 30

FONT_PATH = "assets/fonts/KOMIKAX_.ttf"
FONT_SIZE = 96

TEXT_COLOR = "#FFFFFF"
EMPHASIS_COLOR = "#FFD54F"
STROKE_COLOR = "#000000"
STROKE_WIDTH = 4

PAD_X = 40
PAD_Y = 40
SAFE_TEXT_WIDTH = int(WIDTH * 0.92)

# ---------------- HELPERS ----------------

def load_json(path):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def load_blocks(path):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def get_video_duration(path):
    clip = VideoFileClip(path)
    dur = clip.duration
    clip.close()
    return dur

# ---------------- CAPTION CLIP ----------------

def make_caption_clip(words, start, end, do_pop):
    font = ImageFont.truetype(FONT_PATH, FONT_SIZE)

    dummy = Image.new("RGBA", (10, 10))
    d = ImageDraw.Draw(dummy)

    full_line = " ".join(w["text"] for w in words)
    bbox = d.textbbox((0, 0), full_line, font=font, stroke_width=STROKE_WIDTH)

    text_w = bbox[2] - bbox[0]
    text_h = bbox[3] - bbox[1]

    img = Image.new(
        "RGBA",
        (text_w + PAD_X * 2, text_h + PAD_Y * 2),
        (0, 0, 0, 0),
    )

    draw = ImageDraw.Draw(img)

    x = PAD_X
    y = PAD_Y

    for token in words:
        word = token["text"]
        is_emph = token.get("emphasis", False)
        color = EMPHASIS_COLOR if is_emph else TEXT_COLOR

        draw.text(
            (x, y),
            word + " ",
            font=font,
            fill=color,
            stroke_fill=STROKE_COLOR,
            stroke_width=STROKE_WIDTH,
        )

        x += font.getlength(word + " ")

    clip = ImageClip(np.array(img))

    if clip.w > SAFE_TEXT_WIDTH:
        clip = clip.resized(SAFE_TEXT_WIDTH / clip.w)

    clip = clip.with_position(("center", HEIGHT // 2))

    if do_pop:
        clip = clip.resized(lambda t: 0.85 + min(t / 0.15, 1) * 0.15)

    return clip.with_start(start).with_end(end)

# ---------------- RENDER ----------------

def render_video(audio_path, captions_path, output_path, idea_key):
    print("🎞 Rendering video with MoviePy...")
    os.makedirs(os.path.dirname(output_path), exist_ok=True)

    blocks = load_blocks(captions_path)

    audio = AudioFileClip(audio_path)
    audio_duration = audio.duration

    # ---- LOAD FROZEN IDEA ----
    base_dir = os.path.dirname(os.path.dirname(captions_path))
    frozen_idea_path = os.path.join(base_dir, "input", "idea.json")
    idea_data = load_json(frozen_idea_path)
    idea = idea_data[idea_key]

    gameplay_path = idea["video"]["gameplay"]

    if not os.path.exists(gameplay_path):
        raise FileNotFoundError(f"❌ Gameplay file not found: {gameplay_path}")

    # ---- LOAD GAMEPLAY VIDEO ----
    video_duration = get_video_duration(gameplay_path)
    start_time = random.uniform(0, max(0, video_duration - audio_duration))

    base_video = (
        VideoFileClip(gameplay_path)
        .subclipped(start_time, start_time + audio_duration)
        .resized((WIDTH, HEIGHT))
        .with_fps(FPS)
        .with_audio(audio)
    )

    # ---------------- CAPTIONS ----------------
    caption_clips = []
    prev_end = None

    for block in blocks:
        start = float(block["start"])
        end = float(block["end"])
        words = block["words"]

        do_pop = not (prev_end and start - prev_end < 0.30)

        caption_clips.append(make_caption_clip(words, start, end, do_pop))
        prev_end = end

    # ---------------- OVERLAYS ----------------
    segment_timeline_path = os.path.join(base_dir, "intermediate", "segment_timeline.json")

    overlay_clips = build_overlay_clips(
        word_blocks_path=captions_path,
        segment_timeline_path=segment_timeline_path,
        ideas_path=frozen_idea_path,
        assets_registry_path="assets_registry.json",
        idea_key=idea_key,
    )

    # ---------------- COMPOSE ----------------
    final = CompositeVideoClip(
        [base_video] + overlay_clips + caption_clips,
        size=(WIDTH, HEIGHT),
    )

    final.write_videofile(
        output_path,
        fps=FPS,
        bitrate="20000k",
        codec="libx264",
        audio_codec="aac",
        preset="slow",
        threads=4,
        ffmpeg_params=[
            "-pix_fmt", "yuv420p",
            "-profile:v", "high",
            "-level", "4.2",
            "-movflags", "+faststart"
        ]
    )

    final.close()
    audio.close()
    print("✅ Video rendered:", output_path)

# ---------------- ENTRY ----------------

if __name__ == "__main__":
    if len(sys.argv) != 5:
        print("Usage: python render.py <audio> <captions> <output> <idea_key>")
        sys.exit(1)

    audio_path = sys.argv[1]
    captions_path = sys.argv[2]
    output_path = sys.argv[3]
    idea_key = sys.argv[4]

    render_video(audio_path, captions_path, output_path, idea_key)

# ---------------- QUEUE ----------------

def load_queue():
    if not os.path.exists(QUEUE_FILE):
        return {"last_scheduled_time": None, "queue": []}

    with open(QUEUE_FILE, "r", encoding="utf-8") as f:
        return json.load(f)

def save_queue(data):
    with open(QUEUE_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)

def get_next_schedule_time():
    now = datetime.now(timezone.utc)
    queue = load_queue()

    if queue["last_scheduled_time"]:
        last = datetime.fromisoformat(queue["last_scheduled_time"])
        next_time = last + timedelta(hours=UPLOAD_GAP_HOURS)

        if next_time > now:
            return next_time

    return now

def update_queue(video_path, scheduled_time, youtube_id):
    queue = load_queue()

    queue["last_scheduled_time"] = scheduled_time.isoformat()
    queue["queue"].append({
        "video_path": video_path,
        "scheduled_time": scheduled_time.isoformat(),
        "youtube_id": youtube_id
    })

    save_queue(queue)

# ---------------- AUTH ----------------

def get_authenticated_service():
    creds = None

    if os.path.exists(TOKEN_FILE):
        creds = Credentials.from_authorized_user_file(TOKEN_FILE, SCOPES)

    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        else:
            flow = InstalledAppFlow.from_client_secrets_file(
                CLIENT_SECRETS_FILE, SCOPES
            )
            creds = flow.run_local_server(port=0)

        with open(TOKEN_FILE, "w") as token:
            token.write(creds.to_json())

    return build("youtube", "v3", credentials=creds)

def upload_and_schedule(video_path, title, description, tags):
    if not os.path.exists(video_path):
        raise FileNotFoundError(f"❌ Video file not found: {video_path}")

    youtube = get_authenticated_service()

    scheduled_time = get_next_schedule_time()
    publish_at = scheduled_time.isoformat().replace("+00:00", "Z")

    print("📅 Scheduled time:", scheduled_time.isoformat())

    body = {
        "snippet": {
            "title": title,
            "description": description,
            "tags": tags,
            "categoryId": "27"
        },
        "status": {
            "privacyStatus": "private",
            "publishAt": publish_at,
            "selfDeclaredMadeForKids": False
        }
    }

    media = MediaFileUpload(video_path, chunksize=-1, resumable=True)

    request = youtube.videos().insert(
        part="snippet,status",
        body=body,
        media_body=media
    )

    response = None
    try:
        while response is None:
            status, response = request.next_chunk()
            if status:
                print(f"📤 Uploading: {int(status.progress() * 100)}%")

    except Exception as e:
        msg = str(e)

        if "uploadLimitExceeded" in msg or "exceeded the number of videos" in msg:
            print("🚫 YouTube upload limit reached.")
            print("⏸ Stopping batch safely.")
            sys.exit(99)

        raise

    youtube_id = response["id"]

    update_queue(video_path, scheduled_time, youtube_id)

    print("✅ Uploaded + scheduled")
    print("🆔 Video ID:", youtube_id)
    print("🕒 Scheduled for:", scheduled_time.isoformat())

    return youtube_id

        self.run_btn.clicked.connect(self.start_engine)

        self.log_view = QTextEdit()
        self.log_view.setReadOnly(True)

        layout.addWidget(title)
        layout.addWidget(self.count_label)
        layout.addWidget(self.count_box)
        layout.addWidget(self.run_btn)
        layout.addWidget(self.log_view)

        self.setLayout(layout)

    def start_engine(self):
        self.run_btn.setEnabled(False)
        self.log_view.clear()

        self.worker = EngineWorker(self.count_box.value())
        self.worker.log_line.connect(self.append_log)

        self.worker.failed.connect(self.on_error)
        self.worker.start()

    def append_log(self, text: str):
        self.log_view.append(text)

    def on_success(self):
        self.log_view.append("\n✅ Batch completed successfully.")
        self.run_btn.setEnabled(True)

    def on_error(self, message: str):
        self.log_view.append(f"\n❌ {message}")
        self.run_btn.setEnabled(True)

log_filter.py

NOISE_KEYWORDS = [
    "FutureWarning",
    "UserWarning",
    "TypedStorage",
    "torch._utils",
    "google.api_core",
    "Failed to launch Triton",
    "timing.py"
]

def clean_log(line: str):
    line = line.strip()
    if not line:
        return None

    for key in NOISE_KEYWORDS:
        if key in line:
            return None

    # Friendly wording
    line = line.replace(
        "⚠️ Missing overlay asset:",
        "⚠️ Overlay skipped:"
    )
    line = line.replace(
        "⚠️ Missing SFX in registry:",
        "⚠️ Sound effect skipped:"
    )

    return line

import os
import subprocess
from pathlib import Path

from PySide6.QtCore import QThread, Signal

from app.log_filter import clean_log

# 🔒 Lock engine to your working venv
ENGINE_PYTHON = r"C:\tts\venv\Scripts\python.exe"

class EngineWorker(QThread):
    log_line = Signal(str)
    finished_ok = Signal()
    failed = Signal(str)

    def __init__(self, video_count: int):
        super().__init__()
        self.video_count = video_count

    def run(self):
        try:
            engine_dir = Path(__file__).resolve().parents[2] / "shorts-bot"
            script_path = engine_dir / "generate_batch.py"

            if not script_path.exists():
                self.failed.emit(f"Engine not found at {script_path}")
                return

            # Force UTF-8 for ALL Python subprocesses
            env = os.environ.copy()
            env["PYTHONUTF8"] = "1"

            process = subprocess.Popen(
                [
                    ENGINE_PYTHON,
                    str(script_path),
                    str(self.video_count)
                ],
                cwd=str(engine_dir),
                stdout=subprocess.PIPE,
                stderr=subprocess.STDOUT,
                bufsize=0,          # raw bytes
                env=env
            )

            # 🔑 READ BYTES → DECODE UTF-8 SAFELY
            for raw in iter(process.stdout.readline, b""):
                try:
                    line = raw.decode("utf-8", errors="replace").strip()
                except Exception:
                    continue

                cleaned = clean_log(line)
                if cleaned:
                    self.log_line.emit(cleaned)

            exit_code = process.wait()

            if exit_code == 0:
                self.finished_ok.emit()
            elif exit_code == 99:
                self.failed.emit(
                    "YouTube daily upload limit reached. Resume later."
                )
            else:
                self.failed.emit(
                    "Engine stopped due to an unexpected error."
                )

        except Exception as e:
            self.failed.emit(str(e))
